# set environments
import arcpy
from arcpy.sa import *

arcpy.env.overwriteOutput = False
arcpy.CheckOutExtension("spatial")
arcpy.CheckOutExtension("ImageAnalyst")

# import parameters
DEM = *drag in DEM raster here*
workingGDB = *drag in working GDB here*
dem_extent = arcpy.Raster(DEM).extent

with arcpy.EnvManager(cellSize="1", extent=dem_extent, 
                      scratchWorkspace=workingGDB, 
                      snapRaster=DEM, workspace=workingGDB):
    
    print("Filling Sinks")
    dem2 = arcpy.Raster(DEM)
    elev_fill = arcpy.sa.Fill(dem2, None)

    print("Calculating flow direction...")
    flow_dir = arcpy.sa.FlowDirection(elev_fill, "NORMAL", None, "D8")
    flow_dir_path = workingGDB + "\\FlowDiFill"
    flow_dir.save(flow_dir_path)

    print("Calculating flow accumulation...")
    accum_flow = arcpy.sa.FlowAccumulation(flow_dir, "", "FLOAT", "D8")
    accum_flow_path = workingGDB + "\\AccumFlow"
    accum_flow.save(accum_flow_path)
pourpoint = *drag in pour point here*

# might need to play with this step
with arcpy.EnvManager(cellSize="30", extent=dem_extent, 
                      scratchWorkspace=workingGDB,
                      snapRaster=DEM, workspace=workingGDB):

    print("Snapping pour point...")
    snap_pour_pt = arcpy.sa.SnapPourPoint(pourpoint, accum_flow, 0, "OBJECTID")

    wshed_path = workingGDB + "\\WatershedfromDEM"
    print("Delineating watershed...")
    wshed = arcpy.sa.Watershed(flow_dir, snap_pour_pt, "VALUE")
    wshed.save(wshed_path)
    print("Complete. Watershed output saved to: " + wshed_path)
